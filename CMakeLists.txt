cmake_minimum_required(VERSION 3.1)

project(TOOBAD4ML
    VERSION 0.0.1
    LANGUAGES CXX
)


# Build environment
# -----------------------------------------------------------------------------

# disable in-source builds
if ("${CMAKE_SOURCE_DIR}" STREQUAL "${CMAKE_BINARY_DIR}")
    message(FATAL_ERROR
        "This project does not allow in-source builds. Please remove "
        "CMakeCache.txt and CMakeFiles, then create a separate directory for "
        "the build files."
    )
endif()

# default build type
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug
        CACHE STRING "Default build type set to Debug" FORCE)
endif()

# project directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/lib")

# additional modules
list(APPEND CMAKE_MODULE_PATH      "${CMAKE_CURRENT_SOURCE_DIR}/cmake"
                                   # Conan's autogenerated scripts
                                   "${CMAKE_BINARY_DIR}"
)

# global compiler options
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# user options
option(TOOBAD4ML_ENABLE_TESTING    "Build tests" OFF)
option(TOOBAD4ML_FORCE_CONAN       "Force use dependencies from Conan" OFF)
option(TOOBAD4ML_GENERATE_DOCS "Build documentation" OFF)

# documentation
if (TOOBAD4ML_GENERATE_DOCS)
	find_package(Doxygen)
	if (DOXYGEN_FOUND)
		set(DOXYGEN_IN ${CMAKE_CURRENT_SOURCE_DIR}/docs/Doxyfile)
		set(DOXYGEN_OUT ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile)

		configure_file(${DOXYGEN_IN} ${DOXYGEN_OUT} @ONLY)
		message("DOXYGEN BUILD STARTED. GENERATING DOCS...")

		add_custom_target( doc_doxygen ALL 
			COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYGEN_OUT}
				WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
				COMMENT "Generating API documentation with Doxygen")

	else (DOXYGEN_FOUND)
		message("Doxygen need to be installed to generate the doxygen documentation")
	endif (DOXYGEN_FOUND)
endif (TOOBAD4ML_GENERATE_DOCS)


# Modules
# -----------------------------------------------------------------------------

include(LibTooling)


# Sources
# -----------------------------------------------------------------------------

# source code
add_subdirectory(src)

# test suite
if (TOOBAD4ML_ENABLE_TESTING)
    enable_testing()
    add_subdirectory(tests)
endif(TOOBAD4ML_ENABLE_TESTING)
